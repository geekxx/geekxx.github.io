<div itemscope itemtype="https://schema.org/Recipe">
  <span itemprop="name">Teriyaki Salmon with Broccoli and Jasmine Rice</span>

  <div itemprop="recipeIngredient">
    2 salmon fillets (5–6 oz each)<br>
    ¼ cup teriyaki sauce<br>
    2 cups broccoli florets<br>
    1 cup jasmine rice
  </div>

  <div itemprop="recipeInstructions">
    <ol>
      <li>Cook jasmine rice (1 cup rice + 1¼ cups water).</li>
      <li>Air fry salmon at 400°F for 8–10 minutes, brushing with teriyaki sauce halfway.</li>
      <li>Steam broccoli 4–5 minutes until tender.</li>
    </ol>
  </div>

  <meta itemprop="prepTime" content="PT5M"/>
  <meta itemprop="cookTime" content="PT15M"/>
  <meta itemprop="totalTime" content="PT20M"/>
  <meta itemprop="recipeYield" content="2 servings"/>
</div>
